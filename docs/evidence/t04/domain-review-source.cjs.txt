const assert = require('node:assert/strict');
const fs = require('node:fs');
const flow = require('./t04-domain/onboarding.js');
const rules = require('./t04-domain/placement.js');
const at = '2026-09-12T04:00:00.000Z';
const results = [];
function check(name, run) { run(); results.push({ name, result: 'pass' }); }
function event(state, type, extra = {}) {
  return flow.applyOnboardingEvent(state, { type, questionId: flow.getCurrentQuestion(state)?.id, ...extra });
}
for (const timing of ['after-choice', 'before-choice', 'late-first-playback']) {
  check(`support history ${timing}`, () => {
    let session = flow.createOnboardingSession();
    for (let index = 0; index < 5; index += 1) session = event(session, 'answer', { choiceId: null, response: 'unknown' });
    const question = flow.getCurrentQuestion(session);
    if (timing !== 'late-first-playback') session = event(session, 'playback', { kind: 'prompt', outcome: 'ended', source: 'simulation' });
    if (timing === 'before-choice') session = event(session, 'playback', { kind: 'answer', outcome: 'ended', source: 'simulation' });
    session = event(session, 'answer', { choiceId: question.options[0].id, response: 'choice' });
    session = event(session, 'back');
    session = event(session, 'playback', { kind: 'answer', outcome: 'ended', source: 'simulation' });
    session = event(session, 'playback', { kind: 'prompt', outcome: 'failed', source: 'simulation' });
    session = event(session, 'playback', { kind: 'prompt', outcome: 'ended', source: 'simulation' });
    const snapshot = structuredClone(session);
    const profile = rules.profileFromOnboarding(session, at);
    const support = profile.support_records.find(row => row.question_id === question.id);
    assert.equal(support.answer_before_choice, timing === 'before-choice');
    assert.equal(support.device_failure_before_choice, false);
    assert.equal(support.prompt_played_before_choice, timing !== 'late-first-playback');
    assert.equal(support.answer_exposed, true);
    assert.equal(support.device_failure, true);
    assert.equal(support.prompt_play_count, timing === 'late-first-playback' ? 1 : 2);
    assert.equal(support.replay_count, timing === 'late-first-playback' ? 0 : 1);
    assert.equal(profile.listening_band, null);
    assert.equal(profile.speaking_status, 'unobserved');
    assert.equal(profile.evidence.valid_question_ids.length, 0);
    assert.deepEqual(session, snapshot);
  });
}
const row = (id, band = 'L1', topic = 'daily', patch = {}) => ({ question_id: id, band, topic, task_type: 'intent', tags: ['intent'],
  first_response: 'correct', prompt_played_before_choice: true, device_failure_before_choice: false,
  answer_before_choice: false, translation_before_choice: false, answer_needed: false, replay_count: 0,
  source: 'observed', reviewed: true, local_voice_id: null, local_voice_reviewed: false, is_followup: false,
  recorded_at: at, ...patch });
const evaluate = records => rules.evaluatePlacement(records, { as_of: at, previous_extra_question_ids: [], ended: false });
check('out-of-order repeated snapshots preserve the earliest response', () => {
  const first = row('repeat', 'L1', 'daily', { first_response: 'incorrect' });
  const later = { ...first, first_response: 'correct', recorded_at: '2026-09-12T04:01:00.000Z' };
  const other = row('other', 'L1', 'school');
  const profile = evaluate([later, first, other]);
  assert.equal(profile.listening_band, null);
  assert.deepEqual(profile.evidence.success_question_ids, ['other']);
  assert.equal(profile.evidence.valid_question_ids.length, 2);
});
check('a successful follow-up stays unconfirmed when prerequisites are missing', () => {
  const profile = evaluate([row('l2-a','L2'), row('l2-b','L2','school',{first_response:'incorrect'}),
    row('l2-extra','L2','school',{is_followup:true})]);
  assert.equal(profile.listening_band, null);
  assert.ok(profile.unresolved_bands.includes('L2'));
  assert.deepEqual(profile.followups, []);
});
check('a resolved prerequisite conflict keeps confidence reduced at higher supported bands', () => {
  const records = ['L1','L2','L3'].flatMap(band => [row(`${band}-a`,band),row(`${band}-b`,band,'school')]);
  records.push(row('l2-conflict','L2','work',{first_response:'incorrect'}),row('l2-extra','L2','work',{is_followup:true}));
  const profile = evaluate(records);
  assert.equal(profile.listening_band, 'L3');
  assert.equal(profile.evidence_confidence, 'reduced');
});
check('malformed settings reject null and unsupported schemas without touching their source bytes', () => {
  const profile = evaluate([]);
  for (const raw of ['null','{}','{"schema":9}','{"schema":1,"chosen":null}']) {
    assert.throws(() => rules.readStudySettings(raw,profile));
    assert.equal(typeof raw,'string');
  }
});
const report={operator:'assistant',kind:'supplementary deterministic checks; synthetic observations only',expected:7,passed:results.length,results};
assert.equal(results.length,7);
fs.writeFileSync('docs/evidence/t04/domain-review.json',JSON.stringify(report,null,2)+'\n');
console.log(JSON.stringify({exit:0,passed:results.length}));
